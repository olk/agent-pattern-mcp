def qa_vec(p: dict) -> dict[str, float]:
    return {a: float(p["quality_attributes"].get(a, 0.0)) for a in ATTRS}


def slug_candidates(slugs: list[tuple[str, float]]) -> dict[str, tuple[dict, float]]:
    found: dict[str, tuple[dict, float]] = {}
    for slug, score in slugs:
        norm = slug.lower().replace(" ", "-")
        for p in catalog.values():
            if norm in p.get("suitable_domains", []) and norm not in p.get("unsuitable_domains", []):
                if p["name"] not in found or score > found[p["name"]][1]:
                    found[p["name"]] = (p, score)
    return found


def score_all(cands: dict[str, tuple[dict, float]], w: dict[str, float]) -> list[tuple[str, float]]:
    fused = sorted(cands.values(), key=lambda x: (-x[1], x[0]["name"]))
    f_min = min(x[1] for x in fused)
    f_max = max(x[1] for x in fused)
    f_range = f_max - f_min
    wsum = sum(w.values())
    out = []
    for p, raw in fused:
        v = qa_vec(p)
        analysis = round(10.0 * (sum(w[a] * v[a] for a in ATTRS) / wsum), 2)
        fnorm = 0.0 if f_range == 0 else round((raw - f_min) / f_range * 100.0, 2)
        out.append((p["name"], round(0.7 * analysis + 0.3 * fnorm, 2)))
    out.sort(key=lambda x: -x[1])
    return out


def weight_trials(seed_name: str) -> list[dict[str, float]]:
    v = qa_vec(catalog[seed_name])
    trials: list[dict[str, float]] = []
    t = 1.0
    while t <= 20.0:
        tot = sum(v[a] ** t for a in ATTRS)
        trials.append({a: v[a] ** t / tot for a in ATTRS})
        t += 0.5
    for a in ATTRS:
        oh = {k: 0.0 for k in ATTRS}
        oh[a] = 1.0
        trials.append(oh)
        for b in ATTRS:
            if a < b:
                half = {k: 0.0 for k in ATTRS}
                half[a] = 0.5
                half[b] = 0.5
                trials.append(half)
    rng = random.Random(42)
    for _ in range(800):
        raw = [rng.expovariate(1.0) for _ in ATTRS]
        tot = sum(raw)
        trials.append({a: raw[i] / tot for i, a in enumerate(ATTRS)})
    return trials


def find_weights(seed_name: str, cands: dict[str, tuple[dict, float]]) -> tuple[dict[str, float], float] | None:
    for w in weight_trials(seed_name):
        ranked = score_all(cands, w)
        if ranked[0][0] != seed_name:
            continue
        margin = ranked[0][1] - ranked[1][1]
        if margin > 0.5:
            return (dict(w), margin)
    return None


def main() -> None:
    used: set[tuple[str, str]] = set()
    used_names: set[str] = set()
    scenarios: list[dict] = []
    problems: list[str] = []
    for sid, cat, split, domain, primaries, decoys in PLAN:
        req = REQUIREMENTS[sid]
        assert len(req) >= 40, sid
        prim_cats = {catalog[x]["category"] for x in primaries}
        for d in decoys:
            bad = slug_cats[d] & prim_cats
            if bad:
                problems.append(f"{sid}: decoy {d} leaks {sorted(bad)}")
        dom_order = list(dict.fromkeys(
            [domain]
            + [s for x in primaries for s in catalog[x].get("suitable_domains", [])
               if s not in decoys]
        ))
        chosen: tuple[str, str, dict[str, float], float] | None = None
        fallback: tuple[str, str, dict[str, float], float] | None = None
        for dom in dom_order:
            normal = slug_candidates([(dom, 0.9)] + [(d, 0.3) for d in decoys])
            flipped = slug_candidates([(d, 0.9) for d in decoys] + [(dom, 0.3)])
            on_slug = list(dict.fromkeys(
                [x for x in primaries if dom in catalog[x].get("suitable_domains", [])]
                + [n for n in sorted(catalog)
                   if catalog[n]["category"] == cat and dom in catalog[n].get("suitable_domains", [])]
            ))
            for cand_name in on_slug:
                found = find_weights(cand_name, normal)
                if found is None:
                    continue
                ranked_f = score_all(flipped, found[0])
                if ranked_f[0][0] in primaries:
                    continue
                opt = (cand_name, dom, found[0], found[1])
                if (cand_name, dom) not in used:
                    chosen = opt
                    break
                if fallback is None:
                    fallback = opt
            if chosen is not None:
                break
        if chosen is None and fallback is not None:
            chosen = fallback
        if chosen is None:
            for pname in sorted(catalog):
                if pname in used_names:
                    continue
                for pslug in catalog[pname].get("suitable_domains", []):
                    if pslug in decoys or (pname, pslug) in used:
                        continue
                    normal = slug_candidates([(pslug, 0.9)] + [(d, 0.3) for d in decoys])
                    flipped = slug_candidates([(d, 0.9) for d in decoys] + [(pslug, 0.3)])
                    found = find_weights(pname, normal)
                    if found is None:
                        continue
                    if score_all(flipped, found[0])[0][0] in [pname]:
                        continue
                    chosen = (pname, pslug, found[0], found[1])
                    break
                if chosen is not None:
                    break
        if chosen is None:
            problems.append(f"{sid}: no feasible pattern at all")
            continue
        winner, fdom, weights, margin = chosen
        used.add((winner, fdom))
        used_names.add(winner)
        primaries = [winner] + [x for x in primaries if x != winner][:2]
        fw = {k: round(x, 4) for k, x in weights.items()}
        scenarios.append({
            "scenario_id": sid, "category": cat, "split": split,
            "requirements": req, "domain": fdom,
            "acceptable_primary": primaries, "decoy_domains": decoys,
            "scripted_weights": fw,
        })
        print(f"{sid:36s} w={winner:24s} dom={fdom:26s} margin={margin:5.2f}")

    if problems:
        print("\nPROBLEMS:")
        for p in problems:
            print(" -", p)
        raise SystemExit(1)
    splits = [s["split"] for s in scenarios]
    print(f"\ntrain={splits.count('train')} holdout={splits.count('holdout')}")
    out = REPO / "tests/benchmark/harness/scenarios/seed.json"
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps({"scenarios": scenarios}, indent=2) + "\n")
    print(f"wrote {out}")


if __name__ == "__main__":
    main()
